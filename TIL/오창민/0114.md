## 💻 코드/실습 (Code & Practice)
```Vue
# 예제 코드 작성
<script setup>
import { TresCanvas } from '@tresjs/core'
import { OrbitControls, useGLTF } from '@tresjs/cientos'
import { shallowRef } from 'vue'

const model = shallowRef(null)

// 모델 로드 및 데이터 구조 언래핑 처리
useGLTF('/test.glb').then((result) => {
  let data = result
  
  // AsyncState Wrapper 처리
  if (data.state) data = data.state
  
  // Vue Ref (.value) 처리
  if (data && (data._value !== undefined || data.value !== undefined)) {
    data = data.value
  }

  // 최종 데이터에서 scene 추출
  if (data && (data.scene || data.scenes)) {
    model.value = data.scene || data.scenes[0]
  }
}).catch(err => {
  console.error("Model loading failed:", err)
})
</script>

<template>
  <TresCanvas shadows alpha clear-color="#333">
    <TresPerspectiveCamera :position="[3, 3, 3]" :look-at="[0, 0, 0]" />
    <OrbitControls />
    
    <TresAmbientLight :intensity="3" />
    <TresDirectionalLight :position="[5, 5, 5]" :intensity="3" />

    <primitive 
      v-if="model" 
      :object="model" 
      :scale="20" 
      :position="[0, 0, 0]" 
    />
    
    <TresGridHelper />
    <TresAxesHelper :args="[5]" />
  </TresCanvas>
</template>
```
